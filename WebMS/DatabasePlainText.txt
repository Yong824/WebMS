// Users/User.cs
using Microsoft.EntityFrameworkCore;
using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;
using System.Text.Json.Serialization;
using WebMS.Models.Entities.Commerce;
using WebMS.Models.Entities.Fnbs;
using WebMS.Models.Entities.Movies;
using WebMS.Models.Entities.Places;
using WebMS.Models.Entities.Schedules;
using WebMS.Models.Enums;

#nullable disable warnings

namespace WebMS.Models.Entities.Users;

[Index(nameof(Email), IsUnique = true)]     // For fast lookups, with index SQL Server can lookups for a record by using B-Tree
public class User
{
    [Key]
    public Guid Id { get; set; } = Guid.NewGuid();

    [MaxLength(100)]
    public string Name { get; set; } 

    [MaxLength(20)]
    public string PhoneNumber { get; set; } 

    [MaxLength(255)]
    public string Email { get; set; }

    public bool EmailConfirmed { get; set; } = false;

    /// Hashed password
    [MaxLength(255)]
    [JsonIgnore]
    public string PasswordHash { get; set; } 

    public DateOnly Birthday { get; set; }

    public UserRole Role { get; set; } = UserRole.Customer;

    public UserStatus Status { get; set; } = UserStatus.Normal;

    // When the freeze/block started. 
    public DateTime? FrozenAt { get; set; }

    // When the freeze ends. 
    public DateTime? FrozenUntil { get; set; }

    // ─── Audit ───
    public DateTime CreatedAt { get; set; } = DateTime.UtcNow;
    public DateTime? UpdatedAt { get; set; }
    public DateTime? LastLoginAt { get; set; }

    // ─── Navigation ───
    public List<Transaction> Transactions { get; set; } = [];
    public List<FavoriteMovie> FavoriteMovies { get; set; } = [];
    public List<Movie> Movies { get; set; } = [];               // admin: the cinemas created by this user
    public List<Cinema> Cinemas { get; set; } = [];             // admin: the cinemas created by this user
    public List<Hall> Halls { get; set; } = [];                 // admin: the hall created by this user
    public List<Schedule> Schedules { get; set; } = [];         // admin: the schedule created by this user
    public List<MovieTicket> MovieTickets { get; set; } = [];   // admin: the ticket that scanned by this user, not the ticket owned by the user


    // ─── Computed (NOT columns) ───
    // A frozen user whose FrozenUntil has passed is effectively active again.</summary>
    [NotMapped]
    public bool IsActive =>
        Status == UserStatus.Normal ||
        (Status == UserStatus.Frozen && FrozenUntil.HasValue && FrozenUntil.Value <= DateTime.UtcNow);

    [NotMapped]
    public bool IsAdmin => Role == UserRole.Admin;
}

// MovieFormat.cs
namespace WebMS.Models.Enums;

public enum MovieFormat
{
    Format2D = 0,
    Format3D = 1,
    Imax = 2,
    Format4DX = 3
}

public static class MovieFormatExtensions
{
    public static string ToDisplay(this MovieFormat f) => f switch
    {
        MovieFormat.Format2D => "2D",
        MovieFormat.Format3D => "3D",
        MovieFormat.Imax => "IMAX",
        MovieFormat.Format4DX => "4DX",
        _ => throw new ArgumentOutOfRangeException(nameof(f))
    };

    public static decimal GetDefaultModifier(this MovieFormat format) => format switch
    {
        MovieFormat.Format2D => 0.00m,
        MovieFormat.Format3D => 3.00m,
        MovieFormat.Imax => 10.00m,
        MovieFormat.Format4DX => 15.00m,
        _ => 0.00m
    };
}

namespace WebMS.Models.Enums;

public enum SeatCategory
{
    Standard = 0,
    OKU = 1,
    VIP = 2
}

public static class SeatCategoryExtensions
{
    public static decimal GetDefaultModifier(this SeatCategory category) => category switch
    {
        SeatCategory.Standard => 0.00m,
        SeatCategory.OKU => 0.00m,   // legally should not cost more
        SeatCategory.VIP => 5.00m,

    {
        SeatCategory.Standard => "Standard",

    };

// ShowtimeTier.cs
namespace WebMS.Models.Enums;

public enum ShowtimeTier
{
    Matinee = 0,    // before ~5 PM
    Regular = 1,    // weekday evening
    Prime = 2     // weekend / holiday evening
}

public static class ShowtimeTierExtensions
{
    public static decimal GetDefaultMultiplier(this ShowtimeTier tier) => tier switch
    {
        ShowtimeTier.Matinee => 0.80m,
        ShowtimeTier.Regular => 1.00m,
        ShowtimeTier.Prime => 1.20m,
        _ => 1.00m
    };

    public static string GetDisplayName(this ShowtimeTier tier) => tier switch
    {
        ShowtimeTier.Matinee => "Matinee",
        ShowtimeTier.Regular => "Regular",
        ShowtimeTier.Prime => "Prime (Weekend)",
        _ => tier.ToString()
    };

    /// <summary>Auto-suggests a tier from the showtime. Admin can override.</summary>
    public static ShowtimeTier FromDateTime(DateTime showtime)
    {
        var isWeekend = showtime.DayOfWeek is DayOfWeek.Saturday or DayOfWeek.Sunday;

        if (isWeekend && showtime.Hour >= 17) 
            return ShowtimeTier.Prime;

        if (showtime.Hour < 17) 
            return ShowtimeTier.Matinee;

        return ShowtimeTier.Regular;
    }
}
